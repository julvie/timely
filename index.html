  <script>

    // State management

    let tasks = [

      { id: 1, title: 'Updating Header Text', time: 545, completed: true, running: false },

      { id: 2, title: 'Routine Review Update', time: 545, completed: false, running: false },

      { id: 3, title: 'Working on HTML', time: 545, completed: false, running: false },

      { id: 4, title: 'Working on Email Designs', time: 545, completed: false, running: false }

    ];


    let restMinutes = 30;

    let interval = null;


    // Tab Navigation Switcher

    function switchTab(tabId, btnElement) {

      document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));

      document.querySelectorAll('.nav-btn').forEach(btn => btn.classList.remove('active'));

      

      document.getElementById(tabId).classList.add('active');

      btnElement.classList.add('active');


      if (tabId === 'trendTab') {

        updateTrendMetrics();

      }

    }


    // Format Seconds to HH:MM:SS

    function formatTime(seconds) {

      const h = String(Math.floor(seconds / 3600)).padStart(2, '0');

      const m = String(Math.floor((seconds % 3600) / 60)).padStart(2, '0');

      const s = String(seconds % 60).padStart(2, '0');

      return `${h}:${m}:${s}`;

    }


    // Render Tasks inside Todo Tab

    function renderTasks() {

      const listContainer = document.getElementById('taskList');

      listContainer.innerHTML = '';


      tasks.forEach((task) => {

        const card = document.createElement('div');

        card.className = `task-card ${task.completed ? 'completed' : ''}`;

        

        card.innerHTML = `

          <div class="left-group">

            <span class="drag-handle">=</span

            <div class="checkbox ${task.completed ? 'checked' : ''}" onclick="toggleTask(${task.id})">

              ${task.completed ? '✓' : ''}

            </div>

            <input class="task-title" value="${task.title}" onchange="updateTitle(${task.id}, this.value)">

          </div>

          <div class="right-group">

            <button class="timer-btn ${task.running ? 'running' : ''}" onclick="toggleTimer(${task.id})">

              <span>${task.running ? '❚❚' : '▶'}</span>

              <span>${formatTime(task.time)}</span>

            </button>

            <button class="delete-btn" onclick="deleteTask(${task.id})">🗑</button>

          </div>

        `;

        listContainer.appendChild(card);

      });

    }


    function toggleTask(id) {

      tasks = tasks.map(t => t.id === id ? { ...t, completed: !t.completed } : t);

      renderTasks();

    }


    function updateTitle(id, newTitle) {

      tasks = tasks.map(t => t.id === id ? { ...t, title: newTitle } : t);

    }


    function toggleTimer(id) {

      tasks = tasks.map(t => {

        if (t.id === id) {

          return { ...t, running: !t.running };

        }

        return { ...t, running: false };

      });


      const anyRunning = tasks.some(t => t.running);

      if (anyRunning && !interval) {

        interval = setInterval(() => {

          tasks = tasks.map(t => t.running ? { ...t, time: t.time + 1 } : t);

          renderTasks();

        }, 1000);

      } else if (!anyRunning) {

        clearInterval(interval);

        interval = null;

      }

      renderTasks();

    }


    function addNewTask() {

      tasks.push({ id: Date.now(), title: 'New Task', time: 0, completed: false, running: false });

      renderTasks();

    }


    function deleteTask(id) {

      tasks = tasks.filter(t => t.id !== id);

      renderTasks();

    }


    // Rest activity time tracking

    function addRestTime(minutes) {

      restMinutes += minutes;

      alert(`Added ${minutes} minutes of rest!`);

    }


    // Update Metrics in Trend Tab dynamically

    function updateTrendMetrics() {

      const totalWorkSeconds = tasks.reduce((acc, curr) => acc + curr.time, 0);

      const totalWorkMinutes = Math.max(1, Math.round(totalWorkSeconds / 60));

      

      document.getElementById('totalWorkDisplay').innerText = `${totalWorkMinutes}m`;

      document.getElementById('totalRestDisplay').innerText = `${restMinutes}m`;


      const total = totalWorkMinutes + restMinutes;

      const workPct = Math.round((totalWorkMinutes / total) * 100);

      const restPct = 100 - workPct;


      document.getElementById('workProgressBar').style.width = `${workPct}%`;

      document.getElementById('restProgressBar').style.width = `${restPct}%`;


      document.getElementById('workPercentLabel').innerText = `${workPct}% Work`;

      document.getElementById('restPercentLabel').innerText = `${restPct}% Rest`;


      // Scale today's chart bars relative to current values

      document.getElementById('todayWorkBar').style.height = `${Math.min(100, totalWorkMinutes * 0.8)}px`;

      document.getElementById('todayRestBar').style.height = `${Math.min(100, restMinutes * 0.8)}px`;

    }
